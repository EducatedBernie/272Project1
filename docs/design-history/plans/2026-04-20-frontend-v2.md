"use client";

import { useEffect, useRef } from "react";
import katex from "katex";

export function parseInlineLatex(
  text: string
): { type: "text" | "latex"; content: string }[] {
  const parts: { type: "text" | "latex"; content: string }[] = [];
  const regex = /\$([^$]+)\$/g;
  let lastIndex = 0;
  let match;

  while ((match = regex.exec(text)) !== null) {
    if (match.index > lastIndex) {
      parts.push({ type: "text", content: text.slice(lastIndex, match.index) });
    }
    parts.push({ type: "latex", content: match[1] });
    lastIndex = regex.lastIndex;
  }

  if (lastIndex < text.length) {
    parts.push({ type: "text", content: text.slice(lastIndex) });
  }

  return parts;
}

export function InlineLatex({ latex }: { latex: string }) {
  const ref = useRef<HTMLSpanElement>(null);

  useEffect(() => {
    if (!ref.current) return;
    try {
      katex.render(latex, ref.current, {
        throwOnError: false,
        displayMode: false,
        output: "htmlAndMathml",
      });
    } catch {
      if (ref.current) ref.current.textContent = `$${latex}$`;
    }
  }, [latex]);

  return <span ref={ref} className="mx-0.5" role="math" />;
}

export function DisplayLatex({ latex }: { latex: string }) {
  const ref = useRef<HTMLSpanElement>(null);

  useEffect(() => {
    if (!ref.current) return;
    try {
      katex.render(latex, ref.current, {
        throwOnError: false,
        displayMode: true,
        output: "htmlAndMathml",
      });
    } catch {
      if (ref.current) ref.current.textContent = latex;
    }
  }, [latex]);

  return <span ref={ref} className="block" role="math" />;
}

export function SegmentText({ text }: { text: string }) {
  const parts = parseInlineLatex(text);

  return (
    <span>
      {parts.map((part, i) =>
        part.type === "latex" ? (
          <InlineLatex key={i} latex={part.content} />
        ) : (
          <span key={i}>{part.content}</span>
        )
      )}
    </span>
  );
}
```

import type { Segment, VisualContent, AnchoredSegment } from "./types";

export function anchorVisualsToSegments(
  segments: Segment[],
  visuals: VisualContent[]
): AnchoredSegment[] {
  const sorted = [...segments].sort((a, b) => a.start - b.start);
  const anchored: AnchoredSegment[] = sorted.map((s) => ({ ...s, visuals: [] }));

  for (const v of visuals) {
    let idx = anchored.findIndex(
      (s) => s.start <= v.start && v.start <= s.end
    );
    if (idx === -1) idx = anchored.findIndex((s) => s.start >= v.start);
    if (idx === -1) idx = anchored.length - 1;
    if (idx === -1) continue; // segments is empty; nothing to attach to
    anchored[idx].visuals.push(v);
  }

  for (const s of anchored) s.visuals.sort((a, b) => a.start - b.start);
  return anchored;
}
```

"use client";

import { forwardRef } from "react";
import VideoPlayer, { type PlayerHandle } from "@/components/VideoPlayer";
import { SegmentText, DisplayLatex } from "@/lib/inline-latex";
import type { AnchoredSegment, VideoSource, VisualContent } from "@/lib/types";

interface Section {
  start_time: string;
  end_time?: string;
  title: string;
  description: string;
}

function parseTime(ts: string): number {
  const parts = ts.split(":");
  if (parts.length === 2) return parseInt(parts[0]) * 60 + parseInt(parts[1]);
  if (parts.length === 3)
    return (
      parseInt(parts[0]) * 3600 + parseInt(parts[1]) * 60 + parseInt(parts[2])
    );
  return 0;
}

function formatTime(seconds: number): string {
  const m = Math.floor(seconds / 60);
  const s = Math.floor(seconds % 60);
  return `${m}:${s.toString().padStart(2, "0")}`;
}

function findActiveSegment(
  segments: AnchoredSegment[],
  currentTime: number
): AnchoredSegment | null {
  let best: AnchoredSegment | null = null;
  for (const s of segments) {
    if (s.start <= currentTime) best = s;
    else break;
  }
  return best;
}

function findActiveSection(
  sections: Section[] | null,
  currentTime: number
): { section: Section; index: number } | null {
  if (!sections || sections.length === 0) return null;
  let idx = 0;
  for (let i = sections.length - 1; i >= 0; i--) {
    if (currentTime >= parseTime(sections[i].start_time)) {
      idx = i;
      break;
    }
  }
  return { section: sections[idx], index: idx };
}

function findCurrentVisual(
  segment: AnchoredSegment | null,
  currentTime: number
): VisualContent | null {
  if (!segment || segment.visuals.length === 0) return null;
  let best: VisualContent | null = null;
  for (const v of segment.visuals) {
    if (v.start <= currentTime && currentTime <= v.end) {
      if (!best || v.start > best.start) best = v;
    }
  }
  if (best) return best;
  // fallback: most recent visual that already started
  for (const v of segment.visuals) {
    if (v.start <= currentTime) {
      if (!best || v.start > best.start) best = v;
    }
  }
  return best;
}

interface FocusCardProps {
  source: VideoSource;
  segments: AnchoredSegment[];
  sections: Section[] | null;

"use client";

import { useEffect, useRef } from "react";
import { SegmentText, InlineLatex } from "@/lib/inline-latex";
import type { AnchoredSegment, VisualContent } from "@/lib/types";

interface Section {
  start_time: string;
  end_time?: string;
  title: string;
  description: string;
}

function parseTime(ts: string): number {
  const parts = ts.split(":");
  if (parts.length === 2) return parseInt(parts[0]) * 60 + parseInt(parts[1]);
  if (parts.length === 3)
    return (
      parseInt(parts[0]) * 3600 + parseInt(parts[1]) * 60 + parseInt(parts[2])
    );
  return 0;
}

function formatTime(seconds: number): string {
  const m = Math.floor(seconds / 60);
  const s = Math.floor(seconds % 60);
  return `${m}:${s.toString().padStart(2, "0")}`;
}

interface SegmentListProps {
  segments: AnchoredSegment[];
  sections: Section[] | null;
  currentTime: number;
  onSeek: (time: number) => void;
}

export default function SegmentList({
  segments,
  sections,
  currentTime,
  onSeek,
}: SegmentListProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const activeRef = useRef<HTMLDivElement>(null);

  // Find active segment index
  let activeId: string | null = null;
  for (const s of segments) {
    if (s.start <= currentTime) activeId = s.id;
    else break;
  }

  // Auto-scroll active segment to 1/3 position
  useEffect(() => {
    if (!activeRef.current || !containerRef.current) return;
    const c = containerRef.current;
    const el = activeRef.current;
    const target = el.offsetTop - c.offsetTop - c.clientHeight / 3;
    if (Math.abs(c.scrollTop - target) > 30) {
      c.scrollTo({ top: target, behavior: "smooth" });
    }
  }, [activeId]);

  // Group segments by section
  const sectionStarts = (sections ?? []).map((s, idx) => ({
    ...s,
    index: idx,
    startSeconds: parseTime(s.start_time),
  }));

  function sectionForSegment(segStart: number) {
    if (sectionStarts.length === 0) return null;
    let chosen = sectionStarts[0];
    for (const s of sectionStarts) {
      if (s.startSeconds <= segStart) chosen = s;
      else break;
    }
    return chosen;
  }

  // Walk segments linearly, emit a section header when section changes
  const rows: React.ReactNode[] = [];
  let prevSectionIdx: number | null = null;
  for (const s of segments) {
    const sec = sectionForSegment(s.start);
    if (sec && sec.index !== prevSectionIdx) {
      rows.push(
        <div
          key={`sec-${sec.index}`}
          className="px-5 pt-5 pb-2"
          style={{
            fontFamily: "var(--font-serif)",
            fontStyle: "italic",
            fontSize: 17,
            color: "var(--foreground)",
            borderTop: prevSectionIdx === null ? "none" : "1px solid var(--border)",
          }}
        >
          § {sec.index + 1} · {sec.title}
        </div>
      );
      prevSectionIdx = sec.index;
    }
    const isActive = s.id === activeId;
    rows.push(
      <div
        key={s.id}
        ref={isActive ? activeRef : null}
        className="px-5 py-2.5 cursor-pointer"
        style={{
          background: isActive ? "var(--paper-2)" : "transparent",
          borderLeft: isActive ? "3px solid var(--accent)" : "3px solid transparent",
        }}
        onClick={() => onSeek(s.start)}
        role="button"
        tabIndex={0}
        aria-label={`Segment at ${formatTime(s.start)}: ${s.text}`}
        onKeyDown={(e) => {
          if (e.key === "Enter" || e.key === " ") {
            e.preventDefault();
            onSeek(s.start);
          }
        }}
      >
        <span
          className="text-[11px] font-mono mr-2 align-top"
          style={{ color: "var(--muted-fg)" }}
        >
          {formatTime(s.start)}
        </span>
        <span
          className="text-[15px] leading-[1.55]"
          style={{ color: isActive ? "var(--foreground)" : "var(--muted-fg)" }}
        >
          <SegmentText text={s.text} />
        </span>
        {s.visuals.length > 0 && (
          <div className="mt-2 ml-11 flex flex-col gap-1.5">
            {s.visuals.map((v) => (
              <VisualChip key={v.id} visual={v} onSeek={onSeek} />
            ))}
          </div>
        )}
      </div>
    );
  }

  return (
    <div ref={containerRef} className="h-full overflow-y-auto" style={{ background: "var(--card)" }}>
      {rows}
    </div>
  );
}

function VisualChip({

            height: 20,

            color: fg,
            fontFamily,
            fontStyle: italic ? "italic" : "normal",

          }}

function AutoCaptionPanel({
  captions,
  currentTime,
}: {
  captions: AutoCaption[];
  currentTime: number;
}) {
  const activeRef = useRef<HTMLDivElement>(null);
  const containerRef = useRef<HTMLDivElement>(null);

  let activeIndex = -1;
  for (let i = 0; i < captions.length; i++) {
    if (captions[i].start <= currentTime) activeIndex = i;
    else break;
  }

  useEffect(() => {
    if (activeRef.current && containerRef.current) {
      const c = containerRef.current;
      const el = activeRef.current;
      c.scrollTo({ top: el.offsetTop - c.offsetTop - c.clientHeight / 3, behavior: "smooth" });
    }
  }, [activeIndex]);

  return (
    <div ref={containerRef} className="h-full overflow-y-auto px-3 py-2 space-y-1">
      {captions.map((cap, i) => {
        const isActive = i === activeIndex;
        return (
          <div
            key={i}
            ref={isActive ? activeRef : null}
            className="rounded px-2 py-1.5 transition-all duration-300"
            style={{
              opacity: isActive ? 1 : 0.4,

function buildAnchoredSegments(data: DemoData): AnchoredSegment[] {
  const segments: Segment[] = data.segments.map((s) => ({
    ...s,
    start: Number(s.start),
    end: Number(s.end),
  }));
  const visuals: VisualContent[] = data.visualContent
    ? data.visualContent.map((v) => ({
        ...v,
        start: Number(v.start),
        end: Number(v.end),
        confidence: Number(v.confidence),
      }))
    : data.equations.map((e) => ({
        id: e.id,
        contentType: "equation",
        description: e.description,
        confidence: Number(e.confidence),
        start: Number(e.start),
        end: Number(e.end),
        latex: e.latex,
      }));
  return anchorVisualsToSegments(segments, visuals);
}
